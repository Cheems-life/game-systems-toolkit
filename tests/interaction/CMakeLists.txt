# 独立交互规则与跨模块协调分别链接，分配故障测试放在独立进程。
add_executable(designer_interaction_core_tests core_tests.cpp)
target_link_libraries(designer_interaction_core_tests PRIVATE designer::interaction)
designer_enable_warnings(designer_interaction_core_tests)
add_test(NAME designer.interaction.core COMMAND designer_interaction_core_tests)
foreach(test IN ITEMS player archive allocation configuration)
    add_executable(designer_interaction_${test}_tests ${test}_tests.cpp)
    target_link_libraries(designer_interaction_${test}_tests PRIVATE designer::player designer::configuration)
    target_compile_definitions(designer_interaction_${test}_tests PRIVATE DESIGNER_INTERACTION_FIXTURE="${PROJECT_SOURCE_DIR}/examples/interaction/game.json")
    designer_enable_warnings(designer_interaction_${test}_tests)
    set_target_properties(designer_interaction_${test}_tests PROPERTIES CXX_EXTENSIONS OFF)
    add_test(NAME designer.interaction.${test} COMMAND designer_interaction_${test}_tests)
endforeach()
set_target_properties(designer_interaction_core_tests PROPERTIES CXX_EXTENSIONS OFF)
if(DESIGNER_BUILD_EXAMPLES)
    add_test(NAME designer.interaction.config_check COMMAND designer_config_check ${PROJECT_SOURCE_DIR}/examples/interaction/game.json)
    if(UNIX)
        add_test(NAME designer.interaction_demo.save COMMAND designer_interaction_example ${PROJECT_SOURCE_DIR}/examples/interaction/game.json ${CMAKE_CURRENT_BINARY_DIR}/interaction.sav)
        set_tests_properties(designer.interaction_demo.save PROPERTIES FIXTURES_SETUP saved_interaction)
        add_test(NAME designer.interaction_demo.load COMMAND designer_interaction_example ${PROJECT_SOURCE_DIR}/examples/interaction/game.json ${CMAKE_CURRENT_BINARY_DIR}/interaction.sav --load)
        set_tests_properties(designer.interaction_demo.load PROPERTIES FIXTURES_REQUIRED saved_interaction)
    endif()
endif()
