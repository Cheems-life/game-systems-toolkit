# 任务核心与共享玩家集成分别测试；异常分配器隔离在独立程序，示例以双进程验证恢复。
add_executable(designer_quest_core_tests core_tests.cpp)
target_link_libraries(designer_quest_core_tests PRIVATE designer::quest_core)
set_target_properties(designer_quest_core_tests PROPERTIES CXX_EXTENSIONS OFF)
designer_enable_warnings(designer_quest_core_tests)
add_test(NAME designer.quests.core COMMAND designer_quest_core_tests)
foreach(test IN ITEMS player archive allocation)
    add_executable(designer_quest_${test}_tests ${test}_tests.cpp)
    target_link_libraries(designer_quest_${test}_tests PRIVATE designer::player)
    set_target_properties(designer_quest_${test}_tests PROPERTIES CXX_EXTENSIONS OFF)
    designer_enable_warnings(designer_quest_${test}_tests)
    add_test(NAME designer.quests.${test} COMMAND designer_quest_${test}_tests)
endforeach()

if(DESIGNER_BUILD_EXAMPLES AND UNIX)
    add_test(NAME designer.quests_demo.save COMMAND designer_quests_example ${CMAKE_CURRENT_BINARY_DIR}/player-quests.sav)
    set_tests_properties(designer.quests_demo.save PROPERTIES FIXTURES_SETUP saved_player_quests)
    add_test(NAME designer.quests_demo.load COMMAND designer_quests_example ${CMAKE_CURRENT_BINARY_DIR}/player-quests.sav --load)
    set_tests_properties(designer.quests_demo.load PROPERTIES FIXTURES_REQUIRED saved_player_quests)
endif()
