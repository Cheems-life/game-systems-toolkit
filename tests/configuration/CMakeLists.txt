# 配置输入及真实业务接入回归；JSON 第三方类型不进入测试公共接口。
foreach(test IN ITEMS bundle integration allocation)
    add_executable(designer_config_${test}_tests ${test}_tests.cpp)
    target_link_libraries(designer_config_${test}_tests PRIVATE designer::configuration)
    target_compile_definitions(designer_config_${test}_tests PRIVATE DESIGNER_CONFIG_FIXTURE="${PROJECT_SOURCE_DIR}/examples/configuration/game.json" DESIGNER_CONFIG_SCRATCH="${CMAKE_CURRENT_BINARY_DIR}/oversized.json")
    designer_enable_warnings(designer_config_${test}_tests)
    set_target_properties(designer_config_${test}_tests PROPERTIES CXX_EXTENSIONS OFF)
    add_test(NAME designer.configuration.${test} COMMAND designer_config_${test}_tests)
endforeach()
target_link_libraries(designer_config_integration_tests PRIVATE designer::player)
